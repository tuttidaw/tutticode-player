React

<TuttidawCode> from @tuttidaw/tutticode-player/react (React 17+). A ref gives play(), stop() and setCode(); events are props.

The component

import { useRef, useState } from 'react';
import { TuttidawCode } from '@tuttidaw/tutticode-player/react';

export function Demo() {
  const player = useRef(null);
  const [status, setStatus] = useState('loading');
  return (
    <>
      <TuttidawCode
        ref={player}
        code={'setcpm(20)\ndrums: s("bd ~ [~ bd] ~, ~ sd, hh*8").bank("tr707")'}
        theme="light"
        loops={4}
        onReady={({ title }) => setStatus(`ready: ${title}`)}
        onError={({ code }) => setStatus(`error: ${code}`)}
      />
      <button onClick={() => player.current?.play()}>Play</button>
      <span>{status}</span>
    </>
  );
}
